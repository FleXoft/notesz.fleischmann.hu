<script>
window.addEventListener('DOMContentLoaded', (event) => {
  const rows = document.querySelectorAll('.random-table tr');
  const goldenRatioConjugate = 0.618033988749895;
  let hue = Math.random(); // Kezdőpont véletlenszerű

  rows.forEach((row, index) => {
    if (index === 0) return;

    hue += goldenRatioConjugate;
    hue %= 1; // Megtartjuk a tizedes jegyeket (0 és 1 között)
    
    const finalHue = Math.floor(hue * 360);
    row.style.backgroundColor = `hsl(${finalHue}, 70%, 85%)`;
    row.style.transition = "background-color 0.5s ease";
  });
});</script>
